{{- /* Social link bar. Reads Site.Params.social (map of platform -> value). For platforms with a known URL prefix the value may be just the user handle; full URLs (starting with http/mailto/) and paths (starting with /) are used as-is. `rss: true` links to the site feed. Renders an inline SVG icon per known platform key with optional text labels. Unknown keys fall back to an external-link icon. Pass showLabel=true to render labels. */ -}} {{- $social := .Site.Params.social -}} {{- if not $social }}{{ return }}{{ end -}} {{- $showLabel := .showLabel | default false -}} {{- /* Known URL prefixes per platform. Empty prefix means the value is used verbatim (e.g. mastodon, which needs a full instance URL). */ -}} {{- $prefixes := dict "github" "https://github.com/" "gitlab" "https://gitlab.com/" "twitter" "https://twitter.com/" "x" "https://x.com/" "bluesky" "https://bsky.app/profile/" "linkedin" "https://linkedin.com/in/" "instagram" "https://instagram.com/" "facebook" "https://facebook.com/" "youtube" "https://youtube.com/@" "email" "mailto:" -}}
{{- range $key, $value := $social -}} {{- $key = lower (printf "%v" $key) -}} {{- $url := "" -}} {{- $label := printf "%s%s" (upper (substr $key 0 1)) (substr $key 1) -}} {{- if eq $key "rss" -}} {{- if eq $value true -}} {{- $url = "/feed.xml" | relLangURL -}} {{- else -}} {{- $url = $value -}} {{- end -}} {{- else -}} {{- $v := printf "%v" $value -}} {{- if or (strings.HasPrefix $v "http://") (strings.HasPrefix $v "https://") (strings.HasPrefix $v "mailto:") (strings.HasPrefix $v "/") -}} {{- $url = $v -}} {{- else -}} {{- $prefix := index $prefixes $key -}} {{- if $prefix -}} {{- $url = printf "%s%s" $prefix $v -}} {{- else -}} {{- $url = $v -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq $key "github" -}} {{- else if eq $key "gitlab" -}} {{- else if eq $key "mastodon" -}} {{- else if or (eq $key "x") (eq $key "twitter") -}} {{- else if eq $key "bluesky" -}} {{- else if eq $key "linkedin" -}} {{- else if eq $key "youtube" -}} {{- else if eq $key "instagram" -}} {{- else if eq $key "facebook" -}} {{- else if eq $key "email" -}} {{- else if eq $key "rss" -}} {{- else -}} {{- end -}} {{- if $showLabel -}} {{ $label }} {{- end -}} {{- end -}}
